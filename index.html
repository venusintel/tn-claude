    onValue Event Update Chart

    let tempPoint, humiPoint;
let refTemp = firebase.database().ref("***key ของอุณหภูมิ***");
refTemp.on("value", function (dataSnapshot) {
    let childData = dataSnapshot.val();
    tempPoint = childData.toPrecision(4);
    document.getElementById("***id ของอุณหภูมิ***").innerHTML = tempPoint;
});
let refHumi = firebase.database().ref("***key ของความชื้น***");
refHumi.on("value", function (dataSnapshot) {
    let childData = dataSnapshot.val();
    humiPoint = childData.toPrecision(4);
    document.getElementById("***id ของความชื้น***").innerHTML = humiPoint;
    updateChart();
});
